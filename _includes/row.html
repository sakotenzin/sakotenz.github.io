{%- comment -%} Created by Vitor Venturin on 07/11/2026. Justified media row: every cell gets its aspect ratio, so the row fills the width at equal height. Usage in a post (items separated by "|"): {% include row.html media="photo.jpg 1920x1280 | other.webp 839x596 Alt text here | clip.mp4 1920x1080 clipf001.jpg" %} Image item: x [alt text...] Video item: x {%- endcomment -%}
{%- assign items = include.media | split: "|" -%} {%- for raw in items -%} {%- assign parts = raw | strip | split: " " -%} {%- assign file = parts[0] -%} {%- assign dims = parts[1] | split: "x" -%} {%- assign w = dims[0] | plus: 0 -%} {%- assign h = dims[1] | plus: 0 -%} {%- assign ratio = w | times: 1.0 | divided_by: h | round: 4 -%} {%- assign ext = file | split: "." | last | downcase -%}

{%- endfor -%}
